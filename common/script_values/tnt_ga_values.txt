########################################################################
# Guardian Calculation Assistant 1.0
#
# Pure, GUI-evaluated values. Nothing here writes persistent state.
# Mirrors vanilla education_point_acquisition_support_effect (CK3 1.19,
# common/scripted_effects/00_education_effects.txt): every birthday from
# age 6 to 15 the ward rolls success (+2 points in its focus) vs failure.
# Vanilla factors are referenced by name so overhauls that retune them
# are followed automatically.
#
# Public GUI contract:
#   root               the list row (guardian OR ward candidate)
#   scope:tnt_ga_other the other already-picked secondary character
# The child of the pair is the ward, the adult the guardian. Internal
# values read scope:tnt_ga_guardian and scope:tnt_ga_ward.
########################################################################


########################################################################
# Pair resolution (inlined in each public wrapper below)
########################################################################

tnt_ga_root_is_guardian_trigger_value = {
	value = 0
	if = {
		limit = {
			exists = scope:tnt_ga_other
			this != scope:tnt_ga_other
			is_adult = yes
			scope:tnt_ga_other = { is_adult = no }
		}
		value = 1
	}
}

tnt_ga_root_is_ward_trigger_value = {
	value = 0
	if = {
		limit = {
			exists = scope:tnt_ga_other
			this != scope:tnt_ga_other
			is_adult = no
			scope:tnt_ga_other = { is_adult = yes }
		}
		value = 1
	}
}


########################################################################
# Success weight components
########################################################################

# Guardian's skill in the ward's focus. Without a focus yet, assume the
# guardian's best skill (min = lower bound, i.e. running maximum).
tnt_ga_focus_skill_value = {
	value = 0
	if = {
		limit = { scope:tnt_ga_ward = { has_focus = education_diplomacy } }
		value = scope:tnt_ga_guardian.diplomacy
	}
	else_if = {
		limit = { scope:tnt_ga_ward = { has_focus = education_martial } }
		value = scope:tnt_ga_guardian.martial
	}
	else_if = {
		limit = { scope:tnt_ga_ward = { has_focus = education_stewardship } }
		value = scope:tnt_ga_guardian.stewardship
	}
	else_if = {
		limit = { scope:tnt_ga_ward = { has_focus = education_intrigue } }
		value = scope:tnt_ga_guardian.intrigue
	}
	else_if = {
		limit = { scope:tnt_ga_ward = { has_focus = education_learning } }
		value = scope:tnt_ga_guardian.learning
	}
	else = {
		value = scope:tnt_ga_guardian.diplomacy
		min = scope:tnt_ga_guardian.martial
		min = scope:tnt_ga_guardian.stewardship
		min = scope:tnt_ga_guardian.intrigue
		min = scope:tnt_ga_guardian.learning
	}
}

tnt_ga_ward_has_focus_value = {
	value = 0
	if = {
		limit = { scope:tnt_ga_ward = { has_education_focus_trigger = yes } }
		value = 1
	}
}

# Independent modifiers, exactly as in vanilla.
tnt_ga_ward_intellect_success_value = {
	value = 0
	scope:tnt_ga_ward = {
		if = { limit = { has_trait = intellect_good_3 } add = 20 }
		if = { limit = { has_trait = intellect_good_2 } add = 15 }
		if = { limit = { OR = { has_trait = intellect_good_1 has_trait = shrewd } } add = 10 }
	}
}

tnt_ga_ward_affinity_value = {
	value = 0
	scope:tnt_ga_ward = {
		if = {
			limit = { has_focus = education_diplomacy has_diplomacy_education_affinity_childhood_trait_trigger = yes }
			add = matching_childhood_trait_modifier_value
		}
		if = {
			limit = { has_focus = education_martial has_martial_education_affinity_childhood_trait_trigger = yes }
			add = matching_childhood_trait_modifier_value
		}
		if = {
			limit = { has_focus = education_stewardship has_stewardship_education_affinity_childhood_trait_trigger = yes }
			add = matching_childhood_trait_modifier_value
		}
		if = {
			limit = { has_focus = education_intrigue has_intrigue_education_affinity_childhood_trait_trigger = yes }
			add = matching_childhood_trait_modifier_value
		}
		if = {
			limit = { has_focus = education_learning has_learning_education_affinity_childhood_trait_trigger = yes }
			add = matching_childhood_trait_modifier_value
		}
		if = {
			limit = { has_focus = education_diplomacy has_diplomacy_education_disaffinity_childhood_trait_trigger = yes }
			add = mismatching_childhood_trait_modifier_value
		}
		if = {
			limit = { has_focus = education_martial has_martial_education_disaffinity_childhood_trait_trigger = yes }
			add = mismatching_childhood_trait_modifier_value
		}
		if = {
			limit = { has_focus = education_stewardship has_stewardship_education_disaffinity_childhood_trait_trigger = yes }
			add = mismatching_childhood_trait_modifier_value
		}
		if = {
			limit = { has_focus = education_intrigue has_intrigue_education_disaffinity_childhood_trait_trigger = yes }
			add = mismatching_childhood_trait_modifier_value
		}
		if = {
			limit = { has_focus = education_learning has_learning_education_disaffinity_childhood_trait_trigger = yes }
			add = mismatching_childhood_trait_modifier_value
		}
	}
}

tnt_ga_guardian_intellect_success_value = {
	value = 0
	scope:tnt_ga_guardian = {
		if = { limit = { has_trait = intellect_good_3 } value = 15 }
		else_if = { limit = { has_trait = intellect_good_2 } value = 10 }
		else_if = { limit = { OR = { has_trait = intellect_good_1 has_trait = shrewd } } value = 5 }
	}
}

tnt_ga_guardian_skill_success_value = {
	value = tnt_ga_focus_skill_value
	multiply = educator_skill_modifier_factor
}

tnt_ga_guardian_learning_success_value = {
	value = scope:tnt_ga_guardian.learning
	multiply = educator_learning_skill_modifier_factor
}

# Vanilla writes "value = 10" after the sum, so a storyteller-type culture
# REPLACES intellect + skill + learning with a flat 10. Reproduced as-is.
tnt_ga_guardian_storyteller_value = {
	value = 0
	if = {
		limit = { scope:tnt_ga_guardian.culture = { has_cultural_parameter = better_ward_education } }
		value = 1
	}
}

tnt_ga_guardian_house_value = {
	value = 0
	if = {
		limit = {
			scope:tnt_ga_guardian.house ?= { has_house_aspiration_parameter = service_house_improved_educator_outcome_chance }
		}
		value = 5
	}
}

tnt_ga_success_weight_core_value = {
	value = 0
	add = { value = 60 desc = tnt_ga_base_success }
	if = {
		limit = { tnt_ga_ward_intellect_success_value != 0 }
		add = { value = tnt_ga_ward_intellect_success_value desc = tnt_ga_ward_intellect }
	}
	if = {
		limit = { tnt_ga_ward_affinity_value != 0 }
		add = { value = tnt_ga_ward_affinity_value desc = tnt_ga_ward_affinity }
	}
	if = {
		limit = { tnt_ga_guardian_storyteller_value > 0 }
		add = { value = 10 desc = tnt_ga_guardian_storyteller }
	}
	else = {
		if = {
			limit = { tnt_ga_guardian_intellect_success_value != 0 }
			add = { value = tnt_ga_guardian_intellect_success_value desc = tnt_ga_guardian_intellect }
		}
		add = { value = tnt_ga_guardian_skill_success_value desc = tnt_ga_guardian_skill }
		add = { value = tnt_ga_guardian_learning_success_value desc = tnt_ga_guardian_learning }
	}
	if = {
		limit = { tnt_ga_guardian_house_value != 0 }
		add = { value = tnt_ga_guardian_house_value desc = tnt_ga_guardian_house }
	}
}

	}
}

	value = 0
	scope:tnt_ga_guardian = {

		else_if = { limit = { has_trait = intellect_bad_2 } value = 10 }
		else_if = { limit = { OR = { has_trait = intellect_bad_1 has_trait = dull } } value = 5 }
	}
}

tnt_ga_failure_weight_core_value = {
	value = 0
	add = { value = 40 desc = tnt_ga_base_failure }
	if = {
		limit = { tnt_ga_ward_intellect_failure_value != 0 }
		add = { value = tnt_ga_ward_intellect_failure_value desc = tnt_ga_ward_intellect }
	}
	if = {
		limit = { tnt_ga_guardian_intellect_failure_value != 0 }
		add = { value = tnt_ga_guardian_intellect_failure_value desc = tnt_ga_guardian_intellect }
	}
}

# Percent chance of +2 points per education birthday.
tnt_ga_chance_core_value = {
	value = tnt_ga_success_weight_core_value
	min = 0
	multiply = 100
	divide = {
		value = tnt_ga_success_weight_core_value
		min = 0
		add = tnt_ga_failure_weight_core_value
		min = 1
	}
}

# Birthdays at ages 6..15 still ahead of the ward.
tnt_ga_rolls_left_core_value = {
	value = 15
	subtract = {
		value = scope:tnt_ga_ward.age
		min = 5
	}
	min = 0
}

tnt_ga_current_points_core_value = {
	value = 0
	scope:tnt_ga_ward = {
		if = {
			limit = { has_focus = education_diplomacy has_variable = education_diplomacy_variable }
			value = var:education_diplomacy_variable
		}
		else_if = {
			limit = { has_focus = education_martial has_variable = education_martial_variable }
			value = var:education_martial_variable
		}
		else_if = {
			limit = { has_focus = education_stewardship has_variable = education_stewardship_variable }
			value = var:education_stewardship_variable
		}
		else_if = {
			limit = { has_focus = education_intrigue has_variable = education_intrigue_variable }
			value = var:education_intrigue_variable
		}
		else_if = {
			limit = { has_focus = education_learning has_variable = education_learning_variable }
			value = var:education_learning_variable
		}
	}
}

# Beacon of Learning: at coming of age, +round(points / 4), at most +3.
tnt_ga_beacon_value = {
	value = 0
	if = {
		limit = { scope:tnt_ga_guardian.culture = { has_cultural_parameter = guardian_education_better_outcomes } }
		value = 1
	}
}

# Expected points from guardian rolls only; events, tutors, universities
# and other one-off sources are not predicted.
tnt_ga_projected_points_core_value = {
	value = tnt_ga_current_points_core_value
	add = {
		value = tnt_ga_chance_core_value
		multiply = tnt_ga_rolls_left_core_value
		multiply = 2
		divide = 100
	}
	if = {
		limit = { tnt_ga_beacon_value > 0 }
		save_temporary_value_as = tnt_ga_points_before_beacon
		add = {
			value = scope:tnt_ga_points_before_beacon
			divide = 4
			round = yes
			max = 3
		}
	}
}

# Vanilla cut-offs education_level_1/2/3 (7/12/17). 4 means rank 4 or 5.
tnt_ga_projected_rank_core_value = {
	value = 4
	if = { limit = { tnt_ga_projected_points_core_value <= education_level_1 } value = 1 }
	else_if = { limit = { tnt_ga_projected_points_core_value <= education_level_2 } value = 2 }
	else_if = { limit = { tnt_ga_projected_points_core_value <= education_level_3 } value = 3 }
}

		limit = {
