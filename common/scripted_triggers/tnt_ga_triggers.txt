# Guardian Calculation Assistant: who may be offered as a guardian.
# Mirrors the vanilla educate_child / offer_ward guardian requirements.
tnt_ga_guardian_candidate_trigger = {
	is_alive = yes
	is_physically_able_adult = yes
	is_imprisoned = no
	num_of_relation_ward < 2
}

# Root = candidate; scope:actor = player holding the filter flags.
# No flag set = everyone passes.
tnt_ga_intellect_filter_trigger = {
	OR = {
		scope:actor = {
			NOR = {
				has_variable = tnt_ga_f1
				has_variable = tnt_ga_f2
				has_variable = tnt_ga_f3
			}
		}
		AND = {
			scope:actor = { has_variable = tnt_ga_f1 }
			has_trait = intellect_good_1
		}
		AND = {
			scope:actor = { has_variable = tnt_ga_f2 }
			has_trait = intellect_good_2
		}
		AND = {
			scope:actor = { has_variable = tnt_ga_f3 }
			has_trait = intellect_good_3
		}
	}
}
